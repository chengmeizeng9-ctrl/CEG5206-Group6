cmake_minimum_required(VERSION 3.16)
project(exact_knn_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "Enable address/undefined sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(knn src/knn.cpp src/scalar.cpp src/simd.cpp src/oracle.cpp)
target_include_directories(knn PUBLIC include)
set_source_files_properties(src/scalar.cpp src/oracle.cpp PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize;-fno-tree-slp-vectorize")
set_source_files_properties(src/simd.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-fno-tree-vectorize;-fno-tree-slp-vectorize")
add_executable(knn_tests tests/test.cpp)
target_link_libraries(knn_tests PRIVATE knn)
add_executable(knn_bench src/bench.cpp)
target_link_libraries(knn_bench PRIVATE knn)
enable_testing()
add_test(NAME correctness COMMAND knn_tests)

add_executable(knn_export src/export.cpp)
target_link_libraries(knn_export PRIVATE knn)
